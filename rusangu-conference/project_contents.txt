def create_structure(base="."):
    for root, files in structure.items():
        root_path = os.path.join(base, root)
        os.makedirs(root_path, exist_ok=True)

        for file in files:
            file_path = os.path.join(root_path, file)

            # Create parent directories
            os.makedirs(os.path.dirname(file_path), exist_ok=True)

            # Create empty file if it doesn't exist
            if not os.path.exists(file_path):
                with open(file_path, "w", encoding="utf-8") as f:
                    pass

                print(f"Created: {file_path}")
            else:
                print(f"Exists : {file_path}")

    print("\n===================================")
    print("Project structure created successfully!")
    print("===================================")

        def add_missing_columns(model):
            table_name = model.__tablename__

            existing_columns = {col['name'] for col in inspector.get_columns(table_name)}
            for column in model.__table__.columns:
                if column.name not in existing_columns:
                    col_type = column.type.compile(db.engine)
                    nullable = '' if column.nullable else 'NOT NULL'
                    default = f"DEFAULT {column.default.arg}" if column.default else ''

    __tablename__ = 'users'

    email = db.Column(db.String(120), unique=True, nullable=False)
    password_hash = db.Column(db.String(128), nullable=False)
    first_name = db.Column(db.String(50), nullable=False)
    middle_name = db.Column(db.String(50))
    last_name = db.Column(db.String(50), nullable=False)
    gender = db.Column(db.String(10), nullable=False)
    phone = db.Column(db.String(20), nullable=False)
    diet_preference = db.Column(db.String(50))
    tshirt_size = db.Column(db.String(10))
    tshirt_color = db.Column(db.String(20))
    district_id = db.Column(db.Integer, db.ForeignKey('districts.id'), nullable=True)
    guest_type = db.Column(db.String(20), nullable=False)
    entity = db.Column(db.String(100))
    is_active = db.Column(db.Boolean, default=True)
    created_at = db.Column(db.DateTime, default=datetime.utcnow)
    updated_at = db.Column(db.DateTime, default=datetime.utcnow, onupdate=datetime.utcnow)

    # Specify foreign_keys to resolve ambiguity with approved_by
    user_modules = db.relationship('UserModule', foreign_keys='UserModule.user_id', backref='user', lazy=True, cascade='all, delete-orphan')
    official_designation = db.relationship('OfficialDesignation', backref='user', uselist=False, lazy=True)
    reports_submitted = db.relationship('Report', foreign_keys='Report.submitted_by', backref='submitter', lazy=True)
    reports_reviewed = db.relationship('Report', foreign_keys='Report.reviewed_by', backref='reviewer', lazy=True)

    def set_password(self, password):
        self.password_hash = generate_password_hash(password).decode('utf-8')

    def check_password(self, password):
        return check_password_hash(self.password_hash, password)

    def has_module(self, module_name, active_only=True):
        from app.models.module import Module
        from app.models.user_module import UserModule
        module = Module.query.filter_by(name=module_name).first()
        if not module:
            return False
        q = UserModule.query.filter_by(user_id=self.id, module_id=module.id)
        if active_only:
            q = q.filter_by(is_active=True)
        return q.first() is not None

    def get_modules(self, active_only=True):
        q = UserModule.query.filter_by(user_id=self.id)
        if active_only:
            q = q.filter_by(is_active=True)
        return [um.module.name for um in q.all() if um.module]

from flask import Blueprint, request, jsonify
from flask_jwt_extended import jwt_required, get_jwt_identity
from app import db
from app.models import User, Module, UserModule, District, Department, Report, OfficialDesignation
from app.utils.decorators import require_admin
from sqlalchemy import func

admin_bp = Blueprint('admin', __name__)

@admin_bp.route('/stats', methods=['GET'])
@jwt_required()
@require_admin()
def get_stats():
    total_users = User.query.count()
    total_reports = Report.query.count()
    total_districts = District.query.count()
    total_modules = Module.query.count()
    total_officials = OfficialDesignation.query.count()
    # Reports by status
    reports_by_status = db.session.query(Report.status, func.count(Report.id)).group_by(Report.status).all()
    status_counts = {status: count for status, count in reports_by_status}
    return jsonify({
        'total_users': total_users,
        'total_reports': total_reports,
        'total_districts': total_districts,
        'total_modules': total_modules,
        'total_officials': total_officials,
        'reports_by_status': status_counts
    }), 200

@admin_bp.route('/users', methods=['GET'])
@jwt_required()
@require_admin()
def admin_list_users():
    # list all users with their modules, designation, etc.
    users = User.query.all()
    result = []
    for u in users:
        result.append({
            'id': u.id,
            'email': u.email,
            'name': f"{u.first_name} {u.last_name}",
            'guest_type': u.guest_type,
            'district': u.district.name if u.district else None,
            'is_active': u.is_active,

from flask import Blueprint, request, jsonify
from flask_jwt_extended import create_access_token
from app import db
from app.models import User, District, Module, UserModule

auth_bp = Blueprint('auth', __name__)

@auth_bp.route('/register', methods=['POST'])
def register():
    data = request.get_json()
    required = ['email', 'password', 'first_name', 'last_name', 'gender', 'phone', 'guest_type']
    if not all(k in data for k in required):
        return jsonify({'error': 'Missing required fields'}), 400

    if User.query.filter_by(email=data['email']).first():
        return jsonify({'error': 'Email already registered'}), 400

    if data['guest_type'] not in ['delegate', 'other']:
        return jsonify({'error': 'Invalid guest_type'}), 400

    user = User(
        email=data['email'],
        first_name=data['first_name'],
        middle_name=data.get('middle_name'),
        last_name=data['last_name'],
        gender=data['gender'],
        phone=data['phone'],
        diet_preference=data.get('diet_preference'),
        tshirt_size=data.get('tshirt_size'),
        tshirt_color=data.get('tshirt_color'),
        guest_type=data['guest_type'],
        is_active=True
    )
    user.set_password(data['password'])

    if data['guest_type'] == 'delegate':
        district_id = data.get('district_id')
        if not district_id:
            return jsonify({'error': 'district_id required for delegate'}), 400
        district = District.query.get(district_id)
        if not district:
            return jsonify({'error': 'Invalid district'}), 400
        user.district_id = district_id
        user.entity = None
    else:
        entity = data.get('entity')
        if not entity:
            return jsonify({'error': 'entity required for guest_type other'}), 400
        user.entity = entity
        user.district_id = None

    db.session.add(user)
    db.session.flush()

    # Assign default modules
    use_mod = Module.query.filter_by(name='use').first()
    if use_mod:
        db.session.add(UserModule(user_id=user.id, module_id=use_mod.id, is_active=True))

    if user.guest_type == 'delegate':
        membership_mod = Module.query.filter_by(name='membership').first()
        if membership_mod:
            db.session.add(UserModule(user_id=user.id, module_id=membership_mod.id, is_active=True))
        session_delegate_mod = Module.query.filter_by(name='session_meetings_delegate').first()
        if session_delegate_mod:
            db.session.add(UserModule(user_id=user.id, module_id=session_delegate_mod.id, is_active=False))
    else:
        session_meetings_mod = Module.query.filter_by(name='session_meetings').first()
        if session_meetings_mod:
            db.session.add(UserModule(user_id=user.id, module_id=session_meetings_mod.id, is_active=True))

    db.session.commit()
    return jsonify({'message': 'User registered successfully'}), 201

@auth_bp.route('/login', methods=['POST'])
def login():
    data = request.get_json()
    if not data:
        return jsonify({'error': 'Invalid JSON'}), 400
    email = data.get('email')
    password = data.get('password')
    if not email or not password:
        return jsonify({'error': 'Email and password required'}), 400
    user = User.query.filter_by(email=email).first()
    if not user:
        return jsonify({'error': 'User not found'}), 401
    if not user.check_password(password):
        return jsonify({'error': 'Invalid password'}), 401
    if not user.is_active:
        return jsonify({'error': 'Account disabled'}), 403
    if not user.has_module('use', active_only=True):
        return jsonify({'error': 'No "use" module assigned'}), 403
    # Ensure identity is string
    access_token = create_access_token(identity=str(user.id))
    return jsonify({'access_token': access_token, 'user_id': user.id}), 200

    if designation_type not in valid:
        return jsonify({'error': 'Invalid designation_type'}), 400

    user = User.query.get(user_id)
    if not user:
        return jsonify({'error': 'User not found'}), 404

    # Ensure user has 'officials' and 'reports' modules
    for mod_name in ['officials', 'reports']:
        mod = Module.query.filter_by(name=mod_name).first()
        if mod and not UserModule.query.filter_by(user_id=user_id, module_id=mod.id).first():
            db.session.add(UserModule(user_id=user_id, module_id=mod.id, is_active=True))

    # Check if already designated
    existing = OfficialDesignation.query.filter_by(user_id=user_id).first()
    if existing:
        existing.designation_type = designation_type
        existing.district_id = district_id if designation_type == 'district_pastor' else None
        od = existing
    else:
        od = OfficialDesignation(
            user_id=user_id,
            designation_type=designation_type,
            district_id=district_id if designation_type == 'district_pastor' else None
        )
        db.session.add(od)
        db.session.flush()

    # Manage departments
    # Clear existing
    OfficialDepartment.query.filter_by(official_id=od.id).delete()
    if designation_type == 'director':
        for dept_id in department_ids:
            dept = Department.query.get(dept_id)
            if dept:
                db.session.add(OfficialDepartment(official_id=od.id, department_id=dept_id))
    elif designation_type == 'district_pastor' and department_ids:
        # district pastors can also have departments? The requirement says some can be assigned departments, so we allow.
        for dept_id in department_ids:
            dept = Department.query.get(dept_id)
            if dept:
                db.session.add(OfficialDepartment(official_id=od.id, department_id=dept_id))

    db.session.commit()
    return jsonify({'message': 'Official designated'}), 200

        district_id=data['district_id'],
        quarter=data['quarter'],
        year=data['year'],
        report_type=data['report_type']

    if existing:

        district_id=data['district_id'],
        quarter=data['quarter'],
        year=data['year'],
        report_type=data['report_type'],
        data=data['data'],
        submitted_by=user.id,

from functools import wraps
from flask import jsonify
from flask_jwt_extended import get_jwt_identity, verify_jwt_in_request
from app.models import User

def require_module(module_name, active_only=True):
    def decorator(f):
        @wraps(f)
        def decorated(*args, **kwargs):
            verify_jwt_in_request()
            user_id = get_jwt_identity()
            user = User.query.get(user_id)
            if not user:
                return jsonify({'error': 'User not found'}), 404
            if not user.has_module(module_name, active_only):
                return jsonify({'error': f'Module "{module_name}" required'}), 403
            return f(*args, **kwargs)
        return decorated
    return decorator

def require_official_designation(designation_type):
    def decorator(f):
        @wraps(f)
        def decorated(*args, **kwargs):
            verify_jwt_in_request()
            user_id = get_jwt_identity()
            user = User.query.get(user_id)
            if not user or not user.official_designation:
                return jsonify({'error': 'Not an official'}), 403
            if user.official_designation.designation_type != designation_type:
                return jsonify({'error': f'Designation {designation_type} required'}), 403
            return f(*args, **kwargs)
        return decorated
    return decorator

def require_admin():
    """
    Admin: either has 'admin' module OR is president/executive_secretary.
    """
    def decorator(f):
        @wraps(f)
        def decorated(*args, **kwargs):
            verify_jwt_in_request()
            user_id = get_jwt_identity()
            user = User.query.get(user_id)
            if not user:
                return jsonify({'error': 'User not found'}), 404
            # Check for 'admin' module (active)
            if user.has_module('admin', active_only=True):
                return f(*args, **kwargs)
            # Or official designation president/executive_secretary
            if user.official_designation and user.official_designation.designation_type in ['president', 'executive_secretary']:
                return f(*args, **kwargs)
            return jsonify({'error': 'Admin privileges required'}), 403
        return decorated
    return decorator

import axios from 'axios';

const api = axios.create({
  baseURL: '/api',  // Direct backend URL – no proxy
  headers: {
    'Content-Type': 'application/json',
  },
});

// Response interceptor: handle 401 (excluding login)
api.interceptors.response.use(
  (response) => response,
  (error) => {
    if (error.response?.status === 401 && !error.config.url.includes('/auth/login')) {
      localStorage.removeItem('access_token');
      delete api.defaults.headers.common['Authorization'];
      window.location.href = '/login';
    }
    return Promise.reject(error);
  }
);

export default api;

import React, { useState, useEffect } from 'react';
import api from '../../api/client';

const DepartmentManager = () => {
  const [departments, setDepartments] = useState([]);
  const [newDept, setNewDept] = useState('');

  const fetchDepartments = async () => {
    const res = await api.get('/modules/departments');
    setDepartments(res.data);
  };

  useEffect(() => {
    fetchDepartments();
  }, []);

  const createDept = async () => {
    if (!newDept.trim()) return;
    await api.post('/modules/departments', { name: newDept });
    setNewDept('');
    fetchDepartments();
  };

  const deleteDept = async (id) => {
    if (window.confirm('Delete this department?')) {
      await api.delete(`/modules/departments/${id}`);
      fetchDepartments();
    }
  };

  return (
    <div>
      <h2 className="text-2xl font-bold mb-4">Manage Departments</h2>
      <div className="flex gap-2 mb-4">
        <input type="text" value={newDept} onChange={(e) => setNewDept(e.target.value)} placeholder="New department name" className="border p-2 rounded flex-1" />
        <button onClick={createDept} className="bg-blue-600 text-white px-4 py-2 rounded">Add Department</button>
      </div>
      <ul className="list-disc pl-5">
        {departments.map(d => (
          <li key={d.id} className="flex justify-between items-center border-b py-2">
            <span>{d.name}</span>
            <button onClick={() => deleteDept(d.id)} className="text-red-600">Delete</button>
          </li>
        ))}
      </ul>
    </div>
  );
};

export default DepartmentManager;

import React from 'react';
import { Navigate } from 'react-router-dom';
import { useAuth } from '../../hooks/useAuth';
import { useModules } from '../../hooks/useModules';

const ProtectedRoute = ({ children, requiredModule }) => {
  const { user, loading } = useAuth();
  const { hasModule } = useModules();

  if (loading) return <div className="p-8 text-center">Loading...</div>;
  if (!user) return <Navigate to="/login" />;
  
  if (requiredModule && !hasModule(requiredModule)) {
    return (
      <div className="p-8 text-center">
        <h2 className="text-2xl font-bold text-red-600">Access Denied</h2>
        <p>You do not have the required module: {requiredModule}</p>
      </div>
    );
  }
  return children;
};

const BadgeGenerator = ({ delegates, onUpdate }) => {
  const [selectedIds, setSelectedIds] = useState([]);
  const [generating, setGenerating] = useState(false);

  const toggleSelect = (id) => {
    setSelectedIds(prev =>
      prev.includes(id) ? prev.filter(x => x !== id) : [...prev, id]
    );
  };

  const toggleAll = () => {
    if (selectedIds.length === delegates.length) {
      setSelectedIds([]);
    } else {
      setSelectedIds(delegates.map(d => d.id));
    }
  };

  const generate = async (all = false) => {
    setGenerating(true);
    try {
      await api.post('/delegates/badges/generate', { user_ids: all ? [] : selectedIds });
      onUpdate();
      setSelectedIds([]);
    } catch (error) {
      console.error(error);
    } finally {
      setGenerating(false);
    }
  };

 const downloadBadge = async (userId) => {
  try {
    const response = await api.get(`/delegates/badges/${userId}/download`, {
      responseType: 'blob'
    });
    const url = window.URL.createObjectURL(new Blob([response.data]));
    const link = document.createElement('a');
    link.href = url;
    link.setAttribute('download', `badge_${userId}.pdf`);
    document.body.appendChild(link);
    link.click();
    link.remove();
    window.URL.revokeObjectURL(url);
  } catch (error) {
    console.error('Download failed:', error);
    alert('Failed to download badge. Please try again.');
  }
};

  return (
    <div>
      <div className="flex gap-2 mb-4">
        <button onClick={() => generate(false)} disabled={selectedIds.length === 0 || generating} className="bg-blue-600 text-white px-4 py-2 rounded disabled:opacity-50">
          Generate Selected ({selectedIds.length})
        </button>
        <button onClick={() => generate(true)} disabled={generating} className="bg-green-600 text-white px-4 py-2 rounded disabled:opacity-50">
          Generate All
        </button>
        <button onClick={toggleAll} className="bg-gray-300 px-4 py-2 rounded">
          {selectedIds.length === delegates.length ? 'Deselect All' : 'Select All'}

import React from 'react';
import { useForm } from 'react-hook-form';
import { reportSchemas } from './reportSchemas';

const ReportForm = ({ reportType, initialData, onSubmit }) => {
  const schema = reportSchemas[reportType] || { fields: [] };
  const { register, handleSubmit, formState: { errors } } = useForm({ defaultValues: initialData || {} });

  const renderField = (field) => {
    const common = `border p-2 w-full rounded`;
    if (field.type === 'number') {
      return <input type="number" {...register(field.name, { required: field.required })} className={common} />;
    } else if (field.type === 'text') {
      return <input type="text" {...register(field.name)} className={common} />;
    } else if (field.type === 'textarea') {
      return <textarea {...register(field.name)} className={common} rows="3" />;
    } else if (field.type === 'select') {
      return <select {...register(field.name)} className={common}>
        {field.options.map(opt => <option key={opt} value={opt}>{opt}</option>)}
      </select>;
    }
    return null;
  };

  return (
    <form onSubmit={handleSubmit(onSubmit)} className="space-y-4">
      {schema.fields.map(field => (
        <div key={field.name}>
          <label className="block font-medium">{field.label}</label>
          {renderField(field)}
          {errors[field.name] && <span className="text-red-500 text-sm">This field is required</span>}
        </div>
      ))}
      <button type="submit" className="bg-green-600 text-white px-4 py-2 rounded hover:bg-green-700">Submit</button>
    </form>
  );
};

import React, { createContext, useState, useContext, useEffect } from 'react';
import api from '../api/client';

const AuthContext = createContext();

export const AuthProvider = ({ children }) => {
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    const token = localStorage.getItem('access_token');
    if (token) {
      api.defaults.headers.common['Authorization'] = `Bearer ${token}`;
      fetchUser();
    } else {
      setLoading(false);
    }
  }, []);

  const fetchUser = async () => {
    try {
      const res = await api.get('/users/me');
      setUser(res.data);
    } catch (error) {
      console.error('Fetch user error:', error.response?.data || error.message);
      localStorage.removeItem('access_token');
      delete api.defaults.headers.common['Authorization'];
      setUser(null);
    } finally {
      setLoading(false);
    }
  };

  const login = async (email, password) => {
    const res = await api.post('/auth/login', { email, password });
    const { access_token } = res.data;
    localStorage.setItem('access_token', access_token);
    api.defaults.headers.common['Authorization'] = `Bearer ${access_token}`;
    await fetchUser();
    return res.data;
  };

  const logout = () => {
    localStorage.removeItem('access_token');
    delete api.defaults.headers.common['Authorization'];
    setUser(null);
  };

  return (
    <AuthContext.Provider value={{ user, loading, login, logout }}>
      {children}
    </AuthContext.Provider>
  );
};

export const useAuthContext = () => useContext(AuthContext);

      } finally {
        setLoading(false);
      }
    };

  try {
    const response = await api.get(`/delegates/badges/${info.id}/download`, {

    });

    const link = document.createElement('a');

    link.setAttribute('download', `badge_${info.id}.pdf`);
    document.body.appendChild(link);
    link.click();
    link.remove();
    window.URL.revokeObjectURL(url);
  } catch (error) {
    console.error('Download failed:', error);
    alert('Failed to download badge.');
  }
};

  if (loading) return <div>Loading...</div>;
  if (!info) return <div>Error loading delegate info</div>;

  return (
    <div className="max-w-2xl mx-auto bg-white p-6 rounded shadow">

import React, { useState, useEffect } from 'react';
import { useAuth } from '../hooks/useAuth';
import api from '../api/client';
import ReportForm from '../components/reports/ReportForm';
import ReportList from '../components/reports/ReportList';
import ReportReview from '../components/reports/ReportReview';

const ReportsPage = () => {
  const { user } = useAuth();
  const [reports, setReports] = useState([]);
  const [selectedReport, setSelectedReport] = useState(null);
  const [editing, setEditing] = useState(false);

  const fetchReports = async () => {
    const res = await api.get('/reports/list');
    setReports(res.data);
  };

  useEffect(() => {
    fetchReports();
  }, []);

  const handleSubmit = async (data) => {
    // data includes all fields from form; we need to construct payload
    const payload = {
      district_id: user.district_id || 1, // should be selected in form
      quarter: 1, // could be form field
      year: 2026,
      report_type: 'stewardship', // should be dynamic
      data: data
    };
    await api.post('/reports/', payload);
    fetchReports();
    setEditing(false);
  };

  const handleReview = async (reportId, action) => {
    await api.post(`/reports/${reportId}/review`, { action });
    fetchReports();
  };

  return (
    <div>
      <h1 className="text-2xl font-bold mb-4">Reports</h1>
      {user?.designation && ['district_pastor','director','cfo','executive_secretary','president'].includes(user.designation) && (
        <div className="mb-4">
          <h2 className="text-xl font-semibold">Review Reports</h2>
          <ReportReview reports={reports} onReview={handleReview} />
        </div>
      )}
      <div className="mb-4">
        <button onClick={() => setEditing(!editing)} className="bg-blue-600 text-white px-4 py-2 rounded">
          {editing ? 'Cancel' : 'New Report'}
        </button>
      </div>
      {editing && (
        <div className="bg-white p-4 rounded shadow mb-4">
          <ReportForm reportType="stewardship" onSubmit={handleSubmit} />
        </div>
      )}
      <div>
        <h2 className="text-xl font-semibold mb-2">Your Submitted Reports</h2>
        <ReportList reports={reports} />
      </div>
    </div>
  );
};

export default ReportsPage;
